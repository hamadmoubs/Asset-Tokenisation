// scripts/fix_everything.js
// Run: npx truffle exec scripts/fix_everything.js --network development

const fs = require("fs");
const path = require("path");

const ERC20Token = artifacts.require("ERC20Token");
const KYCWhitelist = artifacts.require("KYCWhitelist"); // adjust name if needed
const Marketplace = artifacts.require("Marketplace");

// where your React _addresses.js lives:
const ADDRESSES_PATH = path.join(__dirname, "..", "src", "_addresses.js");

module.exports = async function (callback) {
  try {
    const networkId = await web3.eth.net.getId();
    const accounts = await web3.eth.getAccounts();
    const deployer = accounts[0];
    const vendor = accounts[1];
    const customer = accounts[2];

    console.log("Network ID:", networkId);
    console.log("Deployer :", deployer);
    console.log("Vendor   :", vendor);
    console.log("Customer :", customer);

    const token = await ERC20Token.deployed();
    const kyc = await KYCWhitelist.deployed();
    const market = await Marketplace.deployed();

    console.log("\nDeployed addresses:");
    console.log("TOKEN_ADDRESS      =", token.address);
    console.log("KYC_ADDRESS        =", kyc.address);
    console.log("MARKETPLACE_ADDRESS=", market.address);

    // 1) Write _addresses.js for the frontend
    const fileContent = `// Auto-generated by scripts/fix_everything.js
// Do not edit manually, redeploy & re-run the script instead.

export const TOKEN_ADDRESS = "${token.address}";
export const KYC_ADDRESS = "${kyc.address}";
export const MARKETPLACE_ADDRESS = "${market.address}";

// Admin address used in AdminDashboard UI (not on-chain enforced)
export const ADMIN_ADDRESS = "${deployer}";
`;

    fs.writeFileSync(ADDRESSES_PATH, fileContent);
    console.log(`\n✅ Wrote addresses to: ${ADDRESSES_PATH}`);

    // 2) Mint tokens to vendor & customer (if possible)
    const amt = web3.utils.toWei("1000", "ether");
    if (typeof token.mint === "function") {
      console.log("\nMinting 1000 tokens to Vendor & Customer from deployer...");
      await token.mint(vendor, amt, { from: deployer });
      await token.mint(customer, amt, { from: deployer });

      const vBal = await token.balanceOf(vendor);
      const cBal = await token.balanceOf(customer);
      console.log("Vendor balance  :", web3.utils.fromWei(String(vBal), "ether"));
      console.log("Customer balance:", web3.utils.fromWei(String(cBal), "ether"));
    } else {
      console.log(
        "\n⚠️ token.mint(...) not found. Skipping minting. Ensure your token has a mint function."
      );
    }

    console.log("\n✅ Fix-everything script completed.");
    callback();
  } catch (err) {
    console.error("\n❌ Fix-everything failed:", err);
    callback(err);
  }
};
